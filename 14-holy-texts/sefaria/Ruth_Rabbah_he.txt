וַיְהִי בִּימֵי שְׁפֹט הַשֹּׁפְטִים <small>(רות א, א)</small>, אוֹי לַדּוֹר שֶׁשָּׁפְטוּ אֶת שׁוֹפְטֵיהֶם, וְאוֹי לַדּוֹר שֶׁשּׁוֹפְטָיו צְרִיכִין לְהִשָּׁפֵט, שֶׁנֶּאֱמַר <small>(שופטים ב, יז)</small>: וְגַם אֶל שֹׁפְטֵיהֶם לֹא שָׁמֵעוּ, וּמִי הָיוּ, רַב אָמַר בָּרָק וּדְבוֹרָה הָיוּ. רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי אָמַר, שַׁמְגַּר וְאֵהוּד הָיוּ. רַב הוּנָא אָמַר דְּבוֹרָה וּבָרָק וְיָעֵל הָיוּ, שְׁפֹט חַד, שֹׁפְטִים תְּרֵין, הַשֹּׁפְטִים תְּלָתָא. 
רַבִּי שָׁאַל לְרַבִּי בְּצַלְאֵל, מַהוּ דִכְתִיב <small>(הושע ב, ז)</small>: כִּי זָנְתָה אִמָּם, אֶפְשָׁר שֶׁשָֹּׂרָה אִמֵּנוּ זוֹנָה הָיְתָה. אָמַר לוֹ חַס וְשָׁלוֹם, אֶלָּא אֵימָתַי דִבְרֵי תוֹרָה מִתְבַּזִּין בִּפְנֵי עַמֵּי הָאָרֶץ בְּשָׁעָה שֶׁבַּעְלֵיהֶן מְבַזִּין אוֹתָם, אֲתָא רַבִּי יַעֲקֹב בַּר אַבְדִימֵי וְעָבְדֵיהּ שְׁמוּעָה, אֵימָתַי דִּבְרֵי תוֹרָה נַעֲשִׂין כְּזוֹנוֹת בִּפְנֵי עַמֵּי הָאָרֶץ, בְּשָׁעָה שֶׁבַּעְלֵיהֶן מְבַזִּין אוֹתָם. רַבִּי יוֹחָנָן מַיְיתֵי לָהּ מֵהָכָא <small>(קהלת ט, טז)</small>: וְחָכְמַת הַמִּסְכֵּן בְּזוּיָה, וְכִי חָכְמָתוֹ שֶׁל רַבִּי עֲקִיבָא שֶׁהָיָה מִסְכֵּן, בְּזוּיָה הָיְתָה, אֶלָּא מַהוּ מִסְכֵּן מִי שֶׁהוּא בָּזוּי בִּדְבָרָיו, כְּגוֹן זָקֵן יוֹשֵׁב וְדוֹרֵשׁ <small>(דברים טז, יט)</small>: לֹא תַטֶּה מִשְׁפָּט, וְהוּא מַטֶּה מִשְׁפָּט. לֹא תַכִּיר פָּנִים, וְהוּא מַכִּיר פָּנִים. לֹא תִקַּח שֹׁחַד, וְהוּא לוֹקֵחַ שֹׁחַד, <small>(שמות כב, כא)</small>: כָּל אַלְמָנָה וְיָתוֹם לֹא תְעַנּוּן, וְהוּא מְעַנֶּה אוֹתָם. שִׁמְשׁוֹן הָלַךְ אַחַר עֵינָיו, שֶׁנֶּאֱמַר <small>(שופטים יד, ג)</small>: אוֹתָהּ קַח לִי כִּי הִיא יָשְׁרָה בְּעֵינָי. גִּדְעוֹן עָבַד עֲבוֹדַת כּוֹכָבִים, שֶׁנֶּאֱמַר <small>(שופטים ח, כז)</small>: וַיַּעַשׂ אוֹתוֹ גִּדְעוֹן לְאֵפוֹד, הֱוֵי שֶׁאֵין לְךָ מִסְכֵּן גָּדוֹל מִזֶּה, אוֹי לַדַּיָּן שֶׁמַּכִּיר פָּנִים בַּמִּשְׁפָּט. תָּנֵי רַבִּי חִיָּא <small>(ויקרא יט, טו)</small>: לֹא תַעֲשׂוּ עָוֶל בַּמִּשְׁפָּט, מְלַמֵּד שֶׁהַדַּיָּן שֶׁמְּקַלְקֵל הַדִּין קָרוּי חֲמִשָּׁה שֵׁמוֹת, עָוֶל, שָׂנאוּי, מְשֻׁקָּץ, חֵרֶם, תּוֹעֵבָה. וְהַקָּדוֹשׁ בָּרוּךְ הוּא קוֹרֵא עָלָיו חֲמִשָּׁה, רָע, מְנָאֵץ, מֵפֵר בְּרִית, מַכְעִיס, וּמַמְרֶה. וְגוֹרֵם חֲמִשָּׁה דְבָרִים לָעוֹלָם, מְטַמֵּא אֶת הָאָרֶץ, וּמְחַלֵּל אֶת הַשֵּׁם, וּמְסַלֵּק אֶת הַשְּׁכִינָה, וּמַפִּיל יִשְׂרָאֵל בַּחֶרֶב, וּמַגְלָם מֵאַרְצָם. אוֹי לַדּוֹר שֶׁמְּקֻלְקָל בְּמִדָּה זוֹ. תָּנֵי רַבִּי חִיָּא <small>(ויקרא יט, לה)</small>: לֹא תַעֲשׂוּ עָוֶל בַּמִּשְׁפָּט בַּדִּין, אִם לַדִּין כְּבָר אָמוּר בַּדִּין, אִם כֵּן לָמָּה נֶאֱמַר בַּמִּשְׁפָּט, בַּמִּדָּה, אֶלָּא מְלַמֵּד שֶׁהַמּוֹדֵד נִקְרָא דַּיָּן, וְאִם שִׁקֵּר קָרוּי חֲמִשָּׁה שֵׁמוֹת, וְגוֹרֵם חֲמִשָּׁה דְבָרִים. אוֹי לַדּוֹר שֶׁמִּדּוֹתָיו שֶׁל שֶׁקֶר, דְּאָמַר רַבִּי בְּנָיָא בְּשֵׁם רַבִּי הוּנָא, אִם רָאִיתָ דּוֹר שֶׁמִּדּוֹתָיו שֶׁל שְׁקָרִים מַלְכוּת בָּאָה וּמִתְגָּרָה בְּאוֹתוֹ הַדּוֹר, מַה טַּעַם <small>(משלי יא, א)</small>: מֹאזְנֵי מִרְמָה תּוֹעֲבַת ה', וּכְתִיב <small>(משלי יא, א)</small>: בָּא זָדוֹן וַיָּבֹא קָלוֹן. אָמַר רַבִּי בֶּרֶכְיָה בְּשֵׁם רַבִּי אַבָּא, כְּתִיב <small>(מיכה ו, יא)</small>: הַאֶזְכֶּה בְּמֹאזְנֵי רֶשַׁע, אֶפְשָׁר כֵּן שֶׁדּוֹר שֶׁמִּדּוֹתָיו שֶׁל שֶׁקֶר זוֹכֶה, אֶלָּא וּבְכִיס אַבְנֵי מִרְמָה. אָמַר רַבִּי לֵוִי אַף משֶׁה רְמָזָהּ לָהֶן לְיִשְׂרָאֵל בַּתּוֹרָה <small>(דברים כה, יג)</small>: לֹא יִהְיֶה לְךָ בְּכִיסְךָ, לֹא יִהְיֶה לְךָ בְּבֵיתְךָ אֵיפָה וְאֵיפָה, וְאִם עָשָׂה כֵן סוֹף שֶׁהַמַּלְכוּת בָּאָה וּמִתְגָּרָה בּוֹ, דִּכְתִיב <small>(דברים כה, טז)</small>: כִּי תוֹעֲבַת ה' אֱלֹהֶיךָ כָּל עֹשֵׂה אֵלֶּה כָּל עֹשֵׂה עָוֶל, וּכְתִיב <small>(דברים כה, יז)</small>: זָכוֹר אֵת אֲשֶׁר עָשָׂה לְךָ עֲמָלֵק בַּדֶּרֶךְ בְּצֵאתְכֶם מִמִּצְרָיִם. 
אָמַר רַבִּי, בְּרָכוֹת מְבָרְכוֹת לְבַעְלֵיהֶן, וּקְלָלוֹת מְקַלְּלוֹת לְבַעְלֵיהֶן. בְּרָכוֹת מְבָרְכוֹת לְבַעְלֵיהֶן, דִּכְתִיב <small>(דברים כה, טו)</small>: אֶבֶן שְׁלֵמָה וָצֶדֶק יִהְיֶה, וְאִם עָשִׂיתָ כֵן, יִהְיֶה לָךְ. וּקְלָלוֹת מְקַלְּלוֹת לְבַעְלֵיהֶן, דִּכְתִיב <small>(דברים כה, יג)</small>: לֹא יִהְיֶה לְךָ בְכִיסְךָ אֶבֶן וָאָבֶן, וְאִם עָשִׂיתָ כֵן, אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא אַתָּה בִּקַּשְׁתָּ לַעֲשׂוֹת גְּדוֹלָה וּקְטַנָּה, חַיֶּיךָ אֲפִלּוּ קְטַנָה אֵין מַסְפִּיקִין בְּיָדוֹ שֶׁל אוֹתוֹ רָשָׁע לַעֲשׂוֹת, דִּכְתִיב <small>(דברים כה, יג)</small>: לֹא יִהְיֶה לְךָ בְּכִיסְךָ. וְדִכְוָותָהּ <small>(שמות כ, יט)</small>: לֹא תַעֲשׂוּן אִתִּי, אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא אַתָּה בִּקַּשְׁתָּ לַעֲשׂוֹת לִי אֱלֹהֵי כֶסֶף וְזָהָב, חַיֶּיךָ שֶׁל עֵץ אֵין מַסְפִּיקִין בְּיָדוֹ שֶׁל אוֹתוֹ רָשָׁע לַעֲשׂוֹת, דִּכְתִיב: לֹא תַעֲשׂוּ לָכֶם. 
וַיְהִי רָעָב בָּאָרֶץ, עֲשָׂרָה רְעָבוֹן בָּאוּ לָעוֹלָם, אֶחָד בִּימֵי אָדָם הָרִאשׁוֹן, אֶחָד בִּימֵי לֶמֶךְ, וְאֶחָד בִּימֵי אַבְרָהָם, וְאֶחָד בִּימֵי יִצְחָק, וְאֶחָד בִּימֵי יַעֲקֹב, וְאֶחָד בִּימֵי אֵלִיָּהוּ, וְאֶחָד בִּימֵי אֱלִישָׁע, וְאֶחָד בִּימֵי דָּוִד, וְאֶחָד בִּימֵי שְׁפֹט הַשֹּׁפְטִים, וְאֶחָד שֶׁמִּתְגַּלְגֵּל וּבָא לָעוֹלָם. אֶחָד בִּימֵי אָדָם הָרִאשׁוֹן, שֶׁנֶּאֱמַר <small>(בראשית ג, יז)</small>: אֲרוּרָה הָאֲדָמָה בַּעֲבוּרֶךָ. וְאֶחָד בִּימֵי לֶמֶךְ, שֶׁנֶּאֱמַר <small>(בראשית ה, כט)</small>: מִן הָאֲדָמָה אֲשֶׁר אֵרַרָה ה'. וְאֶחָד בִּימֵי אַבְרָהָם, שֶׁנֶּאֱמַר <small>(בראשית יב, י)</small>: וַיְהִי רָעָב בָּאָרֶץ וַיֵּרֶד אַבְרָם מִצְרַיְמָה. וְאֶחָד בִּימֵי יִצְחָק, שֶׁנֶּאֱמַר <small>(בראשית כו, א)</small>: וַיְהִי רָעָב בָּאָרֶץ מִלְּבַד הָרָעָב הָרִאשׁוֹן. וְאֶחָד בִּימֵי יַעֲקֹב, שֶׁנֶּאֱמַר <small>(בראשית מה, ו)</small>: כִּי זֶה שְׁנָתַיִם הָרָעָב בְּקֶרֶב הָאָרֶץ. וְאֶחָד בִּימֵי אֵלִיָּהוּ, שֶׁנֶּאֱמַר <small>(מלכים א יז, א)</small>: אִם יִהְיֶה הַשָּׁנִים הָאֵלֶּה טַל וּמָטָר. וְאֶחָד בִּימֵי אֱלִישָׁע, שֶׁנֶּאֱמַר <small>(מלכים ב ו, כה)</small>: וַיְהִי רָעָב גָּדוֹל בְּשֹׁמְרוֹן. וְאֶחָד בִּימֵי דָוִד, שֶׁנֶּאֱמַר <small>(שמואל ב כא, א)</small>: וַיְהִי רָעָב בִּימֵי דָוִד שָׁלשׁ שָׁנִים. וְאֶחָד בִּימֵי שְׁפֹט הַשֹּׁפְטִים, שֶׁנֶּאֱמַר: וַיְהִי רָעָב בָּאָרֶץ. וְאֶחָד שֶׁמִּתְגַּלְגֵּל וּבָא לָעוֹלָם, דִּכְתִיב <small>(עמוס ח, יא)</small>: וְהִשְׁלַחְתִּי רָעָב בָּאָרֶץ לֹא רָעָב לַלֶּחֶם וְלֹא צָמָא לַמַּיִם. רַבִּי הוּנָא בְּשֵׁם רַבִּי שְׁמוּאֵל עִקַּר אַוְותֶנְטִיָּיה שֶׁלָּהֶם הָיָה רָאוּי לָבוֹא בִּימֵי שָׁאוּל וְלֹא הָיָה רָאוּי לָבוֹא בִּימֵי דָוִד, וְעַל יְדֵי שֶׁהָיָה שָׁאוּל גְּרוֹפִית שֶׁל שִׁקְמָה וְלֹא יָכוֹל לַעֲמֹד בּוֹ, גִּלְגְלוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא וֶהֱבִיאוֹ אֵצֶל דָּוִד, שֶׁהוּא גְּרוֹפִית שֶׁל זַיִת וְהָיָה יָכוֹל לַעֲמֹד בּוֹ, מַתְלָא אָמַר שִׁילָה חֲטָא וְיוֹחָנָא מִשְׁתַּלְּמָא, כָּךְ כֻּלָּם לֹא בָאוּ בִּבְנֵי אָדָם שְׁפוּפִין אֶלָּא בִּבְנֵי אָדָם גִּבּוֹרִים שֶׁהָיוּ יְכוֹלִין לַעֲמֹד בָּהֶן. רַבִּי חִיָּא בְּשֵׁם רַבִּי שִׁמְעוֹן בֶּן אֶלְעָזָר, לְזַגָּג שֶׁהָיְתָה בְּיָדוֹ קֻפָּה מְלֵאָה כּוֹסוֹת דְּיוֹטְרִיטוֹן, וְכֵיוָן שֶׁהָיָה מְבַקֵּשׁ לִתְלוֹת קֻפָּתוֹ הָיָה מֵבִיא יָתֵד וְתוֹקְעָהּ וְאַחַר כָּךְ תּוֹלֶה קֻפָּתוֹ. כָּךְ כֻּלָּם לֹא בָאוּ בִּבְנֵי אָדָם שְׁפוּפִין אֶלָּא בִּבְנֵי אָדָם גִּבּוֹרִים. רַבִּי בֶּרֶכְיָה הֲוָה קָרֵי עֲלֵיהוֹן <small>(ישעיה מ, כט)</small>: נֹתֵן לַיָּעֵף כֹּחַ. רַבִּי בֶּרֶכְיָה בְּשֵׁם רַבִּי חֶלְבּוֹ שְׁנַיִם בָּאוּ בִּימֵי אָדָם הָרִאשׁוֹן. רַבִּי הוּנָא בְּשֵׁם רַב אַחָא אֶחָד בִּימֵי אַבְרָהָם וְאֶחָד בִּימֵי לֶמֶךְ. רָעָב שֶׁהָיָה בִּימֵי אֵלִיָּהוּ שֶׁל בַּצֹרֶת הָיָה, שָׁנָה עָבְדַת שָׁנָה לָא עָבְדַת, אֲבָל רָעָב שֶׁבָּא בִּימֵי אֱלִישָׁע, שֶׁל מְהוּמָה הָיָה, שֶׁנֶּאֱמַר <small>(מלכים ב ו, כה)</small>: עַד הֱיוֹת רֹאשׁ חֲמוֹר בִּשְׁמֹנִים כֶּסֶף. רָעָב שֶׁבָּא בִּימֵי שְׁפֹט הַשֹּׁפְטִים, רַבִּי הוּנָא בְּשֵׁם רַבִּי דוֹסָא אַרְבָּעִים וּשְׁתַּיִם סְאוֹת הָיוּ וְנַעֲשׂוּ אַרְבָּעִים וְאַחַת, וְהָא תָּנֵי לֹא יֵצֵא אָדָם לְחוּצָה לָאָרֶץ אֶלָּא אִם כֵּן הָיוּ סָאתַיִם לְקוּחוֹת בְּשֶׁקֶל, אָמַר רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל אֵימָתַי בִּזְּמַן שֶׁאֵינוֹ מוֹצֵא לִקַּח, אֲבָל מוֹצֵא לִקַּח אֲפִלּוּ סְאָה בְּשֶׁקֶל לֹא יֵצֵא יִשְׂרָאֵל חוּצָה לָאָרֶץ. וְהָא תַנְיָא בִּשְׁעַת הַדֶּבֶר בִּשְׁעַת מִלְחָמָה כְּנֹס הָרֶגֶל, וּבִשְׁעַת רְעָבוֹן פַּזֵּר הָרֶגֶל, לָמָּה נֶעֱנַשׁ אֱלִימֶלֶךְ, עַל יְדֵי שֶׁהִפִּיל לִבָּן שֶׁל יִשְׂרָאֵל עֲלֵיהֶם, לְבוּלְיָטִין שֶׁהָיָה שָׁרוּי בִּמְדִינָה, וְהָיוּ בְּנֵי הַמְדִינָה סְבוּרִין עָלָיו וְאוֹמְרִים שֶׁאִם יָבוֹאוּ שְׁנֵי בַּצֹּרֶת הוּא יָכוֹל לְסַפֵּק אֶת הַמְּדִינָה עֶשֶׂר שָׁנִים מָזוֹן, כֵּיוָן שֶׁבָּאת שְׁנַת בַּצֹּרֶת יָצְתָה לָהּ שִׁפְחָתוֹ מְעַיְּלַת בְּסִידְקִי וְקֻפָּתָהּ בְּיָדָהּ, וְהָיוּ בְּנֵי הַמְּדִינָה אוֹמְרִים, זֶהוּ שֶׁהָיִינוּ בְּטוּחִים עָלָיו שֶׁאִם תָּבוֹא בַּצֹּרֶת הוּא יָכוֹל לְפַרְנֵס אוֹתָנוּ עֶשֶׂר שָׁנִים, וַהֲרֵי שִׁפְחָתוֹ עוֹמֶדֶת בְּסִידְקִי וְקֻפָּתָהּ בְּיָדָהּ. כָּךְ אֱלִימֶלֶךְ הָיָה מִגְּדוֹלֵי הַמְּדִינָה וּמִפַּרְנָסֵי הַדּוֹר, וּכְשֶׁבָּאוּ שְׁנֵי רְעָבוֹן אָמַר עַכְשָׁיו כָּל יִשְׂרָאֵל מְסַבְּבִין פִּתְחִי, זֶה בְּקֻפָּתוֹ וְזֶה בְּקֻפָּתוֹ, עָמַד וּבָרַח לוֹ מִפְּנֵיהֶם, הֲדָא הוּא דִכְתִיב: וַיֵּלֶךְ אִישׁ מִבֵּית לֶחֶם יְהוּדָה. 
וַיֵּלֶךְ אִישׁ, גַּרְדּוּם, אֶלָּא בּוֹא וּרְאֵה כַּמָּה חִבֵּב הַקָּדוֹשׁ בָּרוּךְ הוּא בִּיאָתָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל יוֹתֵר מִיצִיאָתָהּ, לְהַלָּן כְּתִיב <small>(עזרא ב, סו)</small>: סוּסֵיהֶם, פִּרְדֵיהֶם, גְּמַלֵּיהֶם, בְּרַם הָכָא כְּתִיב: וַיֵּלֶךְ אִישׁ, גַּרְדוּם, אֶלָּא לְהַלָּן עַל יְדֵי שֶׁהָיוּ יוֹצְאִין מֵהָאָרֶץ לְחוּצָה לָאָרֶץ, לֹא יִחֵס הַכָּתוּב אֶת מָמוֹנָם. וַיֵּלֶךְ אִישׁ גַּרְדּוּם. לָגוּר בִּשְׂדֵה מוֹאָב, אָמַר רַבִּי לֵוִי כָּל מָקוֹם שֶׁאַתָּה מוֹצֵא שָׂדֶה עִיר, עִיר מְדִינָה, מְדִינָה אִפַּרְכְיָא. שָׂדֶה עִיר <small>(מלכים א ב, כו)</small>: עֲנָתֹת לֵךְ עַל שָׂדֶיךָ. עִיר מְדִינָה <small>(יחזקאל ט, ד)</small>: עֲבֹר בְּתוֹךְ הָעִיר בְּתוֹךְ יְרוּשָׁלָיִם. מְדִינָה אִפַּרְכְיָא, שֶׁבַע וְעֶשְׂרִים וּמֵאָה מְדִינָה. הוּא וְאִשְׁתּוֹ וּשְׁנֵי בָנָיו, הוּא עִקָּר, וְאִשְׁתּוֹ טְפֵלָה לוֹ, וּבָנָיו טְפֵלִין לָהֶם.
